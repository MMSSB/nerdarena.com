Lightweight Fetch Wrapper for Next.js

Got tired of writing the same try/catch blocks for API calls. I wrote this tiny utility function that standardizes error handling. Feel free to use it in your projects!

export async function fetcher(url, options = {}) {
  try {
    const res = await fetch(url, options);
    if (!res.ok) throw new Error(res.statusText);
    return await res.json();
  } catch (error) {
    console.error("API Error:", error);
    return null;
  }
}
